The code for the diffusion problem is given here:

import numpy as np
from scipy.integrate import solve_ivp
import matplotlib.pyplot as plt

L = 1  # length of pipe, m
D = 2.0 * 10**-5  # diffusion coefficient, m2/s
dx = 0.01
x_grid = np.arange(0, L + dx, dx)

def rates(t, c, x_grid, dx):
    dcdt = np.zeros(len(x_grid))
    N = len(x_grid) - 1
    for i in range(1, N):
        dcdt[i] = D * (c[i-1] - 2*c[i] + c[i+1]) / dx**2
    return dcdt

tmax = 1000 # seconds
c0 = np.zeros(len(x_grid))
c0[0] = 21

sol = solve_ivp(rates, t_span = [0, tmax], y0 = c0, method = "LSODA",
                t_eval = np.arange(0, tmax, 1), args = (x_grid, dx))

for i in range(0, 100, 10):
    plt.plot(sol.t, sol.y[i,:], label = f'{i} cm')
    plt.legend()
    plt.xlabel('time, s')
    plt.ylabel('conc O2, %')
plt.show()
  1. Try to identify if any boundary or initial conditions are used and where in the code these are implemented?

Solution: When making dcdt inside rates, the change is 0 at both boundaries. The for loop is not doing anything to the boundaries (see range from 1 to and exclusive N), so dcdt at boundries remains 0. This implies that the initial condition at the boundaries will also not change. Hence we have implicitly enforced Dirichlet boundary conditions, with 21% at the left and 0% on the right side of the pipe.

2A Solution: We change the code by removing the diffusion term and adding advection instead. Also change the constants for the problem.

import numpy as np
from scipy.integrate import solve_ivp
import matplotlib.pyplot as plt

L = 10  # new length of pipe, m
v = 0.1 # velocity of water, m/s
dx = 0.01 # step size, m
x_grid = np.arange(0, L + dx, dx)

# first lets try central difference for advection
# and make a dedicated function for that version.
def rates_central(t, c, x_grid, dx):
    dcdt = np.zeros(len(x_grid))
    N = len(x_grid) - 1
    # We can only have 1 boundary condtion, which is the left boundary
    # this means we have to remove the right boundary condition, which      # was a dirichlet before. So we have to rewrite the dcdt[-1], which     # was just set to 0 above and is now incorrect. 
    # We discretize the advection term with a backward difference at the       right boundary, which is a first order derivative.We could not use       central difference here at the boundary, because that would             involve a ghost point that we cannot substitute with a boundary         condition. So we use backward difference instead.
    dcdt[-1] = -v * (c[-1] - c[-2])/dx
    # The boundary to the left is still a Dirchlet and 
    # is handled with the dcdt = np.zeros(len(x_grid)) code already. 
    # now looping over the interior nodes
    for i in range(1, N):
        # central difference discretization of the first order                    derivative (advection is a first order derivative)
        dcdt[i] = - v * (c[i+1] - c[i-1]) / (2*dx)
    return dcdt

# We also make a dedicated function for backward difference for advection

def rates_backward(t, c, x_grid, dx):
    dcdt = np.zeros(len(x_grid))
    N = len(x_grid) - 1
    
    # Now we don't need to take the end boundary out of the for loop
    # because the for loop is also backward difference. We simply extend      the range in the for loop to N+1 and change the code to backward       difference

    for i in range(1, N+1):
        dcdt[i] = - v * (c[i] - c[i-1]) / (dx)
    return dcdt

# simulation parameters: 
tmax = 1000 # seconds
c0 = np.zeros(len(x_grid)) # because we have demi-water to start with (no salt in water)
c0[0] = 35 # g/L, except at left boundary we have the salt water.

# make two separate solver calls using the central and backward rates functions. 
sol_central = solve_ivp(rates_central, t_span = [0, tmax], y0 = c0, method = "LSODA",
                t_eval = np.arange(0, tmax, 1), args = (x_grid, dx))

sol_backward = solve_ivp(rates_backward, t_span = [0, tmax], y0 = c0, method = "LSODA",
                t_eval = np.arange(0, tmax, 1), args = (x_grid, dx))

# Lets plot the concentration of salt at the right end of the pipe over time and for both descretization methods
plt.plot(sol_central.t, sol_central.y[-1,:], label = "central")
plt.plot(sol_backward.t, sol_backward.y[-1,:], label = "backward")
plt.xlabel('time, seconds')
plt.ylabel('concentration, g/L')
plt.legend()
plt.show()

# Here it is clear that the central difference results in unstable solution!

2B Solution: We need to add diffusion term back now.

import numpy as np
from scipy.integrate import solve_ivp
import matplotlib.pyplot as plt

L = 10  # length of pipe, m
D = 1.5 * 10**-9 # m2/s
v = 0.1 # m/s
dx = 0.01 # m
x_grid = np.arange(0, L + dx, dx)

def rates(t, c, x_grid, dx):
    dcdt = np.zeros(len(x_grid))
    N = len(x_grid) - 1
    # we now need two boundary conditions
    # for advection we use backward difference at right boundary as before
    # for diffusion we use a neumann BC with flux = 0 to have the open flow condition
    # the dcdx = 0 at right boundary and using a ghost point in central difference gives 
    # for the diffusion part.  D * (c[-2] - c[-1] + c[-2])/dx**2
    # therefore the right end is handled like this:
    dcdt[-1] = -v * (c[-1] - c[-2])/dx + D * (c[-2] - 2* c[-1] + c[-2])/dx**2

    for i in range(1, N):
        # we learned that backward difference got stable results for advection
        # so we keep that. Then we add diffusion with central difference
        dcdt[i] = - v * (c[i] - c[i-1]) / (dx) + D * (c[i+1] - 2*c[i] + c[i-1])/dx**2
    return dcdt

tmax = 1000 # seconds
c0 = np.zeros(len(x_grid))
c0[0] = 35 # g/L

sol = solve_ivp(rates, t_span = [0, tmax], y0 = c0, method = "LSODA",
                t_eval = np.arange(0, tmax, 1), args = (x_grid, dx))

# lets plot the concentration of salt at the right end of the pipe over time
# and for both descretization methods
plt.plot(sol.t, sol.y[-1,:], label = "adv + dif")
plt.plot(sol_backward.t, sol_backward.y[-1, :], label = "adv")
plt.xlabel('time, seconds')
plt.ylabel('concentration, g/L')
plt.legend()
plt.show()

# they are identical! So diffusion does not really matter in this problem.
