Time-variable model solution

Instead of the normal qmd file, let’s set this all up with scripts. We will do parameter estimation in a single script, which will include the definition of a function for calculating residuals as well as the least_squares() call to do the parameter estimation. This present file will display the script contents. You can download the files directly from https://github.com/AU-BCE-EE/course-modelling-2026/tree/main/classes/22-12-nov-time-var/solutions.

Here is the time-variable model module.

benz_modv.py
"""
File name: benz_modv.py
Author: Sasha D. Hafner
Course: Modelling 2026

Description:
    This module defines numerical models for benzene volatilization
    from a wastewater lagoon. The version here includes a time-variable
    input.

Usage:
    See time_var_solution.qmd.
"""

# Load packages 
import numpy as np
from scipy.integrate import solve_ivp
from scipy.interpolate import PchipInterpolator

def benzv(clin, cl0, ca, wind_dat, depth, h, hrt, t_range, t_step):  

    """ 
    Dynamic numerical model for volatilization of benzene 
    from a lagoon, as a batch system.

    Parameters
    ----------
    clin : float
        Benzene concentration in untreated wastewater (g/m3) 
    cl0 : float
        Initial benzene concentration in lagoon (g/m3) 
    ca : float
        Constant ambient benzene concentration in air (g/m3)
    wind_dat : DataFrame
        Pandas DataFrame with time (s) in the first column and 
        wind speed at 2 m height above the lagoon surface (m/s)
        in the second column
    depth : float
        Lagoon depth (assumed to be uniform) (m)
    h : float
        Henry's law constant (volumetric concentration based, aq:g)
    hrt : float
        hydraulic retention time based on flow rate (in = out) and lagoon volume (s)
    t_range : list or tuple of two floats 
        Minimum and maximum time in output (s) 
    t_step : float
        Time step in output (s)

    Returns
    -------
    dictionary
        With elements 't' for time (s) and 'cl' for benzene concentration 
        in wastewater (g/m3)
 
    """

    # Create interpolation function for wind speed
    wind_func = PchipInterpolator(wind_dat.iloc[:, 0],  wind_dat.iloc[:, 1], extrapolate = False)
     
    # Define rates function
    def rates(t, cl):

        # Calculate wind speed (m/s)
        ws = wind_func(t)
        # And mass transfer coefficient (m/s)
        kg =  1e-6 * (1 + 0.1 * ws**2) / (1 + 0.1)

        # Finally, derivative (g/m3-s)
        dcdt = (clin - cl) / hrt - kg * (cl / h - ca) / depth

        return dcdt

    res = solve_ivp(
        rates, 
        t_span = t_range, 
        y0 = [cl0], 
        t_eval = np.arange(t_range[0], t_range[1] + t_step, t_step)
    )

    # Return user-friendly results object (dictionary here)
    out = {
        "t": res.t, 
        "cl": res.y[0, :]
    }

    return out

And here is the application of the model. This is a script.

time_var_benz.py
"""
File: time_var_benzene.py
Class: Modelling 2026
Author: Sasha D. Hafner

Description:
    A script that demonstrates a version of the benzene volatilization
    model with a time-variable input.
"""

# Load modules
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.integrate import solve_ivp

# The model module
import benz_modv as bmv

# The measurement data
weather = pd.read_csv('weather.csv')

# The model function docstring says we need time in s in the first column and wind speed in m/s in second
# Best option may be to make a new DataFrame
wind = pd.DataFrame(dict(time_s = weather['time_hr'] * 3600, wind_m_s = weather['wind_m_s']))

# Let's start from a high (fresh) benzene concentration . . .
pred_high = bmv.benzv(
    clin=10, 
    cl0=10, 
    ca=0, 
    wind_dat=wind, 
    depth=1,
    h=5,
    hrt=21 * 86400,
    t_range = (0, 40 * 86400),
    t_step = 3600
)

# and a low one
pred_low = bmv.benzv(
    clin=10, 
    cl0=0, 
    ca=0, 
    wind_dat=wind, 
    depth=1,
    h=5,
    hrt=21 * 86400,
    t_range = (0, 40 * 86400),
    t_step = 3600
)

# And make some higher wind speed just for fun
wind2 = pd.DataFrame(dict(time_s = weather['time_hr'] * 3600, wind_m_s = 5 * weather['wind_m_s']))

pred_windy = bmv.benzv(
    clin=10, 
    cl0=3, 
    ca=0, 
    wind_dat=wind2, 
    depth=1,
    h=5,
    hrt=21 * 86400,
    t_range = (0, 40 * 86400),
    t_step = 3600
)


plt.plot(pred_low['t'] / 86400, pred_low['cl'], label = 'Low start')
plt.plot(pred_high['t'] / 86400, pred_high['cl'], label = 'High start')
plt.plot(pred_windy['t'] / 86400, pred_windy['cl'], label = 'Windy, high start')
plt.xlabel('Time (days)')
plt.ylabel('Benzene conc. (g/m3)')
plt.savefig('benz1.png')
